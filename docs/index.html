      # grab a release binary, or build
$ curl -sL https://github.com/nestho/bbmonitor/releases/latest/download/bbmonitor_linux_amd64 -o bbmonitor
$ chmod +x bbmonitor && ./bbmonitor -once

$ BB_MODE=daemon BB_INTERVAL=10m ./bbmonitor
